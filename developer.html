Code Examples

Engine Initialization

EngineInitialization.cs
91 lines
using Resonex.Audio;
using Resonex.Config;
using UnityEngine;

public class AudioManager : MonoBehaviour
{
    [SerializeField] private int maxAudioSources = 128;
    [SerializeField] private bool enableHRTF = true;
    [SerializeField] private bool enableRoomSimulation = true;
    [SerializeField] private QualityLevel qualityLevel = QualityLevel.High;
    
    private ISpatialAudioEngine engine;
    
    void Awake()
    {
        // Create configuration
        var config = new SpatialAudioConfig
        {
            MaxAudioSources = maxAudioSources,
            MaxActiveSources = 32,
            HRTFEnabled = enableHRTF,
            HRTFDataset = HRTFDataset.Default,
            RoomSimulation = enableRoomSimulation,
            RayTracingEnabled = false,
            QualityLevel = qualityLevel,
            SampleRate = 48000,
            BufferSize = AudioBufferSize.Medium,
            SpatializerPlugin = "Resonex",
            
            // Performance settings
            EnableThreading = true,
            WorkerThreadCount = SystemInfo.processorCount - 1,
            EnableSIMD = true,
            
            // Debug settings
            DebugMode = DebugMode.Release,
            EnableProfiling = false,
            
            // Callbacks
            OnSourceCreated = HandleSourceCreated,
            OnSourceDestroyed = HandleSourceDestroyed,
            OnOcclusionCalculated = HandleOcclusion
        };
        
        // Initialize engine
        engine = SpatialAudioEngine.Initialize(config);
        
        if (engine == null)
        {
            Debug.LogError("Failed to initialize Resonex Audio Engine");
            return;
        }
        
        // Configure listener
        engine.SetListenerPosition(transform.position);
        engine.SetListenerRotation(transform.rotation);
        
        Debug.Log($"Resonex Audio Engine initialized: v{engine.GetVersion()}");
    }
    
    void HandleSourceCreated(ISpatialAudioSource source)
    {
        Debug.Log($"Audio source created: {source.Id}");
    }
    
    void HandleSourceDestroyed(int sourceId)
    {
        Debug.Log($"Audio source destroyed: {sourceId}");
    }
    
    void HandleOcclusion(int sourceId, float occlusionFactor)
    {
        // Custom occlusion handling
    }
    
    void Update()
    {
        // Update listener transform
        engine.SetListenerPosition(transform.position);
        engine.SetListenerRotation(transform.rotation);
        engine.SetListenerVelocity(rigidbody?.velocity ?? Vector3.zero);
        
        // Per-frame update
        engine.Update(Time.deltaTime);
    }
    
    void OnDestroy()
    {
        engine?.Shutdown();
    }
}
